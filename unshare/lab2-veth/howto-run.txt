CONTAINER_PID=12345
# 宿主机网卡名称（通过 ip route 或 ip a 查看，例如 eth0 或 wlan0）
HOST_IF="eth0"


1. 创建 veth 网卡对，并将一端放入容器中：

# 创建一对虚拟网卡：veth-host 和 veth-guest
ip link add veth-host type veth peer name veth-guest

# 将 veth-guest 移入容器的网络命名空间
ip link set veth-guest netns $CONTAINER_PID


2. 配置宿主机端的 veth 网卡：

# 为宿主机的 veth 端分配 IP 并启动
ip addr add 192.168.100.1/24 dev veth-host
ip link set veth-host up


3. 开启宿主机的 IP 转发与 NAT 转发规则：

# 开启内核 IP 转发功能
sysctl -w net.ipv4.ip_forward=1

# 配置 iptables 进行地址伪装 (NAT)，允许容器流量通过宿主机外网网卡出去
iptables -t nat -A POSTROUTING -s 192.168.100.0/24 -o $HOST_IF -j MASQUERADE
iptables -A FORWARD -i veth-host -o $HOST_IF -j ACCEPT
iptables -A FORWARD -i $HOST_IF -o veth-host -m state --state RELATED,ESTABLISHED -j ACCEPT



步骤 3：在容器内部配置 IP 地址与默认网关（在容器终端执行）


# 启动本地回环网卡 lo
ip link set lo up

# 重命名 veth 网卡为 eth0（可选，方便识别）
ip link set veth-guest name eth0

# 给容器网卡配置 IP
ip addr add 192.168.100.2/24 dev eth0
ip link set eth0 up

# 添加默认路由，将流量指向宿主机的 veth 地址
ip route add default via 192.168.100.1
